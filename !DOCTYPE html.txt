let teacherPollTimer = null;

function switchTab(name){
  document.getElementById('view-student').classList.toggle('active', name === 'student');
  document.getElementById('view-teacher').classList.toggle('active', name === 'teacher');
  document.getElementById('tabBtnStudent').classList.toggle('active', name === 'student');
  document.getElementById('tabBtnTeacher').classList.toggle('active', name === 'teacher');

  if(name === 'teacher'){
    loadAll();
    if(!teacherPollTimer){
      teacherPollTimer = setInterval(() => loadAll({ silent:true }), 15000); // 15초마다 자동으로 제출 현황 새로고침
    }
  } else if(teacherPollTimer){
    clearInterval(teacherPollTimer);
    teacherPollTimer = null;
  }
}

  const onTeacherTab = document.getElementById('view-teacher').classList.contains('active');
  if(document.hidden){
    if(teacherPollTimer){ clearInterval(teacherPollTimer); teacherPollTimer = null; }
  } else if(onTeacherTab){
    loadAll({ silent:true });
    if(!teacherPollTimer) teacherPollTimer = setInterval(() => loadAll({ silent:true }), 15000);
  }
});

}

async function submitForm(){
  const regionA = document.getElementById('regionA').value.trim();
  const regionB = document.getElementById('regionB').value.trim();
  const hyp = document.getElementById('hypothesis').value.trim();
  const dNature = document.getElementById('dataNature').value.trim();
  const dCulture = document.getElementById('dataCulture').value.trim();
  const dPopulation = document.getElementById('dataPopulation').value.trim();
  const dIndustry = document.getElementById('dataIndustry').value.trim();
  const conclusion = document.getElementById('conclusion').value.trim();
  const reflection = document.getElementById('reflection').value.trim();

  const filledDataCount = [dNature, dCulture, dPopulation, dIndustry].filter(v => v.length > 0).length;

  if(!regionA || !regionB || !hyp){
    alert('① 가설 단계 정보가 비어 있어요. 다시 확인해주세요.');
    goStep(1);
    return;
  }
  if(filledDataCount < 2){
    alert('② 조사 데이터를 2가지 이상 작성해주세요.');
    goStep(2);
    return;
  }
  if(!conclusion){
    showError('최종결론을 입력해주세요.');
    return;
  }
  hideError();

  if(SCRIPT_URL.indexOf('여기에') !== -1){
    showError('선생님이 아직 웹 앱 주소를 연결하지 않았어요. 잠시만 기다려주세요.');
    return;
  }

  const btn = document.getElementById('submitBtn');
  btn.disabled = true;
  btn.innerHTML = '<span class="spinner"></span>제출 중...';

  const row = [studentNumber, studentName, regionA, regionB, hyp, dNature, dCulture, dPopulation, dIndustry, conclusion, reflection, nowStr()];

  try{
    const res = await fetch(SCRIPT_URL, {
      method:'POST',
      body: JSON.stringify({ sheet:'학생응답', row })
    });
    const result = await res.json().catch(() => null);

    if(!result || result.ok !== true){
      showError('저장에 실패했어요. (' + (result && result.error ? result.error : '알 수 없는 오류') + ') 선생님께 알려주세요.');
      btn.disabled = false;
      btn.textContent = '제출하기';
      return;
    }

    formSubmitted = true;
    document.getElementById('formCard').style.display = 'none';
    document.getElementById('successCard').style.display = 'block';

    const rev = document.getElementById('reviewBox');
    rev.innerHTML = `

async function loadAll(opts){
  const silent = !!(opts && opts.silent);

  if(!dashboardInited){
    document.getElementById('classSizeLabel').textContent = CLASS_SIZE;
    dashboardInited = true;
  }

  // 자동 새로고침(폴링) 중에 선생님이 피드백을 입력하고 있으면 건너뛰어 내용이 사라지지 않게 함
  if(silent){
    const active = document.activeElement;
    if(active && (active.classList.contains('fb-input') || active.classList.contains('score-select'))){
      return;
    }
  }

  const body = document.getElementById('tableBody');
  if(SCRIPT_URL.indexOf('여기에') !== -1){
    body.innerHTML = '<tr><td colspan="8" class="empty">Apps Script 웹앱 URL이 아직 연결되지 않았습니다. 상단 코드의 SCRIPT_URL을 채워주세요.</td></tr>';
    return;
  }
  if(!silent){
    body.innerHTML = '<tr><td colspan="8" class="empty">불러오는 중...</td></tr>';
  }
  try{
    [submissionRows, feedbackRows] = await Promise.all([
      fetchSheet('학생응답'),
      fetchSheet('피드백')
    ]);
    renderStats();
    renderTable();
    const lu = document.getElementById('lastUpdated');
    if(lu) lu.textContent = '· 마지막 업데이트 ' + timeOnly();
  }catch(err){
    if(!silent){
      body.innerHTML = '<tr><td colspan="8" class="empty">데이터를 불러오지 못했습니다. URL과 시트 이름을 확인해주세요.</td></tr>';
    }
  }
}

function latestFeedbackFor(number){
  const mine = feedbackRows.filter(r => String(r[0]).trim() === String(number).trim());
  return mine.length ? mine[mine.length - 1] : null;
}

function renderStats(){
  const submittedNumbers = new Set(submissionRows.map(r => String(r[0]).trim()));
  document.getElementById('totalSubmitted').textContent = submittedNumbers.size;
  const missing = Math.max(CLASS_SIZE - submittedNumbers.size, 0);
  document.getElementById('totalMissing').textContent = missing;
  document.getElementById('totalFeedback').textContent = feedbackRows.length;

  const missingBox = document.getElementById('missingBox');
  if(missing > 0){
    missingBox.style.display = 'block';
    missingBox.textContent = `아직 제출하지 않은 학생이 ${missing}명 있습니다. (전체 ${CLASS_SIZE}명 중 ${submittedNumbers.size}명 제출)`;
  } else {
    missingBox.style.display = 'none';
  }
}

    const fb = latestFeedbackFor(number);

    const tr = document.createElement('tr');

      <td>${escapeHtml(name)}</td>

      <td class="cell-wide">${escapeHtml(hyp)}</td>
      <td class="cell-wide">${[
        dNature ? `<b>자연</b> ${escapeHtml(dNature)}` : '',
        dCulture ? `<b>문화</b> ${escapeHtml(dCulture)}` : '',
        dPopulation ? `<b>인구</b> ${escapeHtml(dPopulation)}` : '',

      ].filter(Boolean).join('<br>')}</td>
      <td class="cell-wide">${escapeHtml(conclusion)}</td>
      <td class="cell-wide">${escapeHtml(reflection)}</td>
      <td>
        <textarea class="fb-input" id="fbInput-${idx}" placeholder="피드백을 입력하세요">${fb ? escapeHtml(fb[2]) : ''}</textarea>
        <select class="score-select" id="fbScore-${idx}">
          <option value="상" ${fb && fb[4]==='상' ? 'selected':''}>상</option>
          <option value="중" ${fb && fb[4]==='중' ? 'selected':''}>중</option>
          <option value="하" ${fb && fb[4]==='하' ? 'selected':''}>하</option>
        </select>
        <div class="row-actions">
          <button class="btn small ghost" onclick="generateAIDraft(${idx}, '${escapeAttr(hyp)}', '${escapeAttr(dNature)}', '${escapeAttr(dCulture)}', '${escapeAttr(dPopulation)}', '${escapeAttr(dIndustry)}', '${escapeAttr(conclusion)}')">AI 초안 생성</button>
          <button class="btn small" onclick="saveFeedback(${idx}, '${escapeAttr(number)}', '${escapeAttr(name)}')">저장</button>
        </div>
        <div class="saved-tag" id="saved-${idx}">저장되었습니다</div>
      </td>
    `;
    body.appendChild(tr);
  });
}

function escapeHtml(v){
  return String(v ?? '').replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
}
function escapeAttr(v){
  return String(v ?? '').replace(/'/g, "\\'").replace(/\n/g, ' ');
}

async function saveFeedback(idx, number, name){
  const content = document.getElementById(`fbInput-${idx}`).value.trim();
  const score = document.getElementById(`fbScore-${idx}`).value;
  if(!content){ alert('피드백 내용을 입력해주세요.'); return; }

  const writer = content.startsWith('[AI 초안]') ? 'AI(교사검토)' : '교사';
  const row = [number, name, content, writer, score, nowStr()];

  await fetch(SCRIPT_URL, {
    method:'POST',
    body: JSON.stringify({ sheet:'피드백', row })
  });

  const tag = document.getElementById(`saved-${idx}`);
  tag.style.display = 'block';
  setTimeout(() => tag.style.display = 'none', 2500);
  feedbackRows.push(row);
  document.getElementById('totalFeedback').textContent = feedbackRows.length;
}

async function generateAIDraft(idx, hyp, dNature, dCulture, dPopulation, dIndustry, conclusion){
  const key = document.getElementById('geminiKey').value.trim();
  const textarea = document.getElementById(`fbInput-${idx}`);
  textarea.value = '초안 생성 중...';

  if(key){
    try{
      const prompt = `너는 고등학교 지리 교사다. 학생의 탐구 내용을 보고 학생에게 건네는 피드백을 작성해라.
가설: ${hyp}
자연환경 조사: ${dNature || '(작성 안 함)'}
문화 조사: ${dCulture || '(작성 안 함)'}
인구 조사: ${dPopulation || '(작성 안 함)'}
산업 조사: ${dIndustry || '(작성 안 함)'}
최종결론: ${conclusion}
조건: 잘한 점 1가지와 보완할 점 1가지를 따뜻한 존댓말로, 60자 이내로 작성. 정답을 대신 알려주지 말고 스스로 더 생각해볼 질문을 포함할 것.`;

      const res = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.0-flash:generateContent?key=${key}`, {
        method:'POST',
        headers:{'Content-Type':'application/json'},
        body: JSON.stringify({ contents:[{ parts:[{ text: prompt }] }] })
      });
      const data = await res.json();
      const text = data?.candidates?.[0]?.content?.parts?.[0]?.text;
      textarea.value = '[AI 초안] ' + (text ? text.trim() : '생성에 실패했습니다. 직접 작성해주세요.');
    }catch(err){
      textarea.value = '[AI 초안] 생성 중 오류가 발생했습니다. 직접 작성해주세요.';
    }
  } else {
    textarea.value = `[AI 초안] 자연환경과 문화를 연결지어 설명한 점이 좋습니다. "${conclusion.slice(0,20)}..." 부분은 구체적인 근거를 한 가지 더 들면 더 탄탄해질 것 같아요. 다른 계절이나 지형에서도 같은 결론이 유지될까요?`;
  }
}

function downloadCSV(){
  if(submissionRows.length === 0){ alert('다운로드할 데이터가 없습니다.'); return; }
  const header = ['학번','이름','비교지역A','비교지역B','가설','조사데이터_자연환경','조사데이터_문화','조사데이터_인구','조사데이터_산업','최종결론','활동소감','제출시각'];
  const lines = [header.join(',')];
  submissionRows.forEach(row => {
    lines.push(row.map(v => `"${String(v ?? '').replace(/"/g,'""')}"`).join(','));
  });
  const blob = new Blob(["\uFEFF" + lines.join('\n')], { type:'text/csv;charset=utf-8;' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = '학생응답_' + nowStr().slice(0,10) + '.csv';
  a.click();
  URL.revokeObjectURL(url);
}
